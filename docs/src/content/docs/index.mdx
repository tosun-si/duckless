import { Card, CardGrid } from '@astrojs/starlight/components';

```bash
uv tool install duckless
duckless init --project my-project --region europe-west1
duckless run daily_revenue.sql --machine n2-highmem-32 --spot
```

<CardGrid stagger>
	<Card title="Your project, your data" icon="seti:lock">

	</Card>
	<Card title="GCS through ADC" icon="approve-check">
		The runner reads and writes `gs://` with the VM's service account. No HMAC keys, no JSON key files.

	<Card title="Big machines, short lives" icon="rocket">
		Pick an `n2-highmem-32` for five minutes, with local SSD for spill and Spot pricing. The VM is deleted when the job ends.

	<Card title="SQL or your own code" icon="seti:python">
